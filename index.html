Instant monetization

This is your entire paywall logic:

import { init, isPaid, showPaywall } from "sdk-2minpay";

// Pass your email
await init("developer@example.com");

if (!await isPaid()) {
  // Monthly subscription
  await showPaywall(2.99, true);
  
  // Or one-time payment
  await showPaywall(29.99, false);
}
